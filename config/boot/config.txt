###############################################################################
# Raspberry Pi Firmware Configuration
# PI-DMX CONTROLLER v2
#
# This is the complete /boot/firmware/config.txt for the DMX controller.
# Copy this file to /boot/firmware/config.txt after a fresh OS install.
#
# Hardware:
# - USB audio input (no HiFiBerry HAT)
# - SPI OLED display (EastRising SSD1322 256×64, CE0)
# - RS485 DMX output via UART (/dev/serial0)
# - 5 rotary encoders (GPIO)
# - Bluetooth disabled to free UART for DMX
###############################################################################

# -----------------------
# Enable required buses
# -----------------------

# SPI for OLED display (spidev0.0 only). Single chip-select (CE0/GPIO8) so that
# CE1/GPIO7 is NOT reserved by the SPI driver and is free for the extra button.
dtoverlay=spi0-1cs

# I2C (optional peripherals)
dtparam=i2c_arm=on

# Enable UART (for DMX over RS485)
enable_uart=1

# Disable Bluetooth to free primary UART for DMX
dtoverlay=disable-bt


# -----------------------
# Audio
# -----------------------

# USB capture — no HiFiBerry overlay. Onboard analog disabled (USB is separate card).
dtparam=audio=off


# -----------------------
# Display / GPU
# -----------------------

# Enable DRM / KMS graphics
dtoverlay=vc4-kms-v3d
max_framebuffers=2

# Let kernel decide video mode
disable_fw_kms_setup=1

# Disable overscan compensation
disable_overscan=1


# -----------------------
# Camera / display autodetect
# -----------------------

camera_auto_detect=1
display_auto_detect=1


# -----------------------
# Initramfs
# -----------------------

auto_initramfs=1


# -----------------------
# Performance
# -----------------------

arm_boost=1


# -----------------------
# Compute Module support
# -----------------------

[cm4]
otg_mode=1

[cm5]
dtoverlay=dwc2,dr_mode=host

[all]

# SPI single chip-select is configured above (dtoverlay=spi0-1cs): the OLED uses
# CE0/GPIO8, leaving CE1/GPIO7 free for the extra button. Do NOT use
# `dtparam=spi=on` (the 2-CS default), which reserves GPIO7 and breaks that button.
